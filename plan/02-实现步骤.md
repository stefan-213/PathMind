```typescript
import { create } from 'zustand';

interface ChatMessage {
  id: string;
  role: 'user' | 'assistant';
  content: string;
  hasGraph?: boolean;
  timestamp: number;
}

interface ChatState {
  messages: ChatMessage[];
  sessionId: string | null;
  isLoading: boolean;

  addMessage: (message: Omit<ChatMessage, 'id' | 'timestamp'>) => void;
  setSessionId: (sessionId: string) => void;
  setLoading: (loading: boolean) => void;
  clearMessages: () => void;
}

export const useChatStore = create<ChatState>((set) => ({
  messages: [],
  sessionId: null,
  isLoading: false,

  addMessage: (message) => set((state) => ({
    messages: [...state.messages, {
      ...message,
      id: crypto.randomUUID(),
      timestamp: Date.now()
    }]
  })),

  setSessionId: (sessionId) => set({ sessionId }),
  setLoading: (isLoading) => set({ isLoading }),
  clearMessages: () => set({ messages: [], sessionId: null }),
}));
```

### Step 4：创建通用图表渲染组件（4 小时）

**文件：** `src/pages/Graph/GraphCanvas.tsx`

核心要点：
1. 根据 `data.layout` 选择 D3 布局策略
2. 根据 `data.type` 应用不同渲染样式
3. 支持拖拽、缩放交互

```typescript
import { useEffect, useRef } from 'react';
import * as d3 from 'd3';
import type { GraphData, GraphNode } from '../../types/graph';

interface GraphCanvasProps {
  data: GraphData | null;
  onNodeClick?: (node: GraphNode) => void;
}

export function GraphCanvas({ data, onNodeClick }: GraphCanvasProps) {
  const svgRef = useRef<SVGSVGElement>(null);
  const containerRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    if (!svgRef.current || !containerRef.current || !data) return;

    const svg = d3.select(svgRef.current);
    const container = containerRef.current;
    const width = container.clientWidth;
    const height = container.clientHeight;

    // 清空
    svg.selectAll('*').remove();

    // 创建映射
    const nodeMap = new Map(data.nodes.map(n => [n.id, n]));

    // D3 数据
    const d3Nodes = data.nodes.map((n) => ({
      ...n,
      x: width / 2 + (Math.random() - 0.5) * 200,
      y: height / 2 + (Math.random() - 0.5) * 200,
    }));

    const d3Links = data.edges
      .filter(e => nodeMap.has(e.source) && nodeMap.has(e.target))
      .map(e => ({ ...e }));

    const g = svg.append('g');

    // 缩放
    const zoom = d3.zoom<SVGSVGElement, unknown>()
      .scaleExtent([0.2, 4])
      .on('zoom', (event) => g.attr('transform', event.transform));
    svg.call(zoom);

    // 根据 layout 选择布局
    let simulation;
    switch (data.layout) {
      case 'tree':
        simulation = createTreeLayout(d3Nodes, d3Links, width, height);
        break;
      case 'timeline':
        simulation = createTimelineLayout(d3Nodes, d3Links, width, height);
        break;
      default:
        simulation = createForceLayout(d3Nodes, d3Links, width, height);
    }

    // 渲染
    renderLinks(g, d3Links);
    renderNodes(g, d3Nodes, onNodeClick);

    simulation.on('tick', () => updatePositions(d3Links, d3Nodes));

    return () => simulation.stop();
  }, [data, onNodeClick]);

  return (
    <div className="graph-canvas" ref={containerRef}>
      <svg ref={svgRef} className="graph-svg" />
      {!data && <EmptyState />}
    </div>
  );
}
```

### Step 5：创建 AI 对话面板（3 小时）

**文件：** `src/pages/Graph/AISidePanel.tsx`

核心要点：
1. 管理对话历史
2. 调用 AI 接口
3. 解析 AI 返回的图表数据并触发更新

```typescript
export function AISidePanel({ onGraphUpdate }: AISidePanelProps) {
  const { messages, addMessage, setLoading, sessionId } = useChatStore();
  const [input, setInput] = useState('');

  const handleSend = async () => {
    if (!input.trim() || !sessionId) return;

    const userMessage = input.trim();
    setInput('');
    addMessage({ role: 'user', content: userMessage });
    setLoading(true);

    try {
      const res = await chatWithAI({
        agentId: '100001',
        sessionId,
        message: userMessage
      });

      const response = res.data;
      const aiContent = response.message || '';
      const graphData = response.graphData;

      addMessage({
        role: 'assistant',
        content: aiContent,
        hasGraph: !!graphData
      });

      // 如果 AI 返回了图表数据，更新图表
      if (graphData) {
        onGraphUpdate(graphData);
      }
    } catch (err) {
      addMessage({
        role: 'assistant',
        content: '抱歉，发生了错误'
      });
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="ai-side-panel">
      <div className="messages-container">
        {/* 消息列表 */}
      </div>
      <div className="input-area">
        <textarea
          value={input}
          onChange={(e) => setInput(e.target.value)}
          onKeyDown={(e) => {
            if (e.key === 'Enter' && !e.shiftKey) {
              e.preventDefault();
              handleSend();
            }
          }}
        />
        <button onClick={handleSend}>发送</button>
      </div>
    </div>
  );
}
```

```typescript
export function GraphPage() {
  const { graphData, setGraphData } = useGraphStore();
  const { sessionId } = useChatStore();
  const [isPanelOpen, setIsPanelOpen] = useState(true);

  // 初始化会话
  useEffect(() => {
    const initSession = async () => {
      const res = await createSession('100001');
      useChatStore.getState().setSessionId(res.data?.sessionId);
    };
    initSession();
  }, []);

  return (
    <div className="graph-page">
      <Header />
      <div className="graph-main">
        <div className={`graph-canvas-area ${isPanelOpen ? 'with-panel' : ''}`}>
          <GraphCanvas data={graphData} />
        </div>
        {isPanelOpen && (
          <AISidePanel onGraphUpdate={setGraphData} />
        )}
        <button onClick={() => setIsPanelOpen(!isPanelOpen)}>
          {isPanelOpen ? '关闭面板' : '打开面板'}
        </button>
      </div>
      <Footer />
    </div>
  );
}
```

```java
@RestController
@RequestMapping("/api/agent")
public class AIController {

    @PostMapping("/chatStream")
    public ApiResponse<AIChatResponse> chat(@RequestBody ChatRequest request) {
        // 调用 AI，传入对话历史
        String aiResponse = aiService.chat(
            request.getMessage(),
            request.getSessionId()
        );

        // 解析 AI 返回内容
        AIChatResponse response = parseAIResponse(aiResponse);

        return ApiResponse.success(response);
    }

    @PostMapping("/createSession")
    public ApiResponse<Map<String, String>> createSession(@RequestBody Map<String, String> request) {
        String sessionId = aiService.createSession(request.get("agentId"));
        return ApiResponse.success(Map.of("sessionId", sessionId));
    }
}
```
