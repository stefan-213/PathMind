# 优化方案：AI 智能图表生成与对话系统

## 一、性能优化

### 1.1 大图表渲染优化

**问题：** 当图表节点超过 100 个时，D3 SVG 渲染会变慢

**解决方案：** 节点超过阈值时切换到 Canvas 模式

```typescript
const USE_CANVAS_THRESHOLD = 100;

function GraphCanvas({ data }: GraphCanvasProps) {
  const [renderMode, setRenderMode] = useState<'svg' | 'canvas'>('svg');

  useEffect(() => {
    if (data && data.nodes.length > USE_CANVAS_THRESHOLD) {
      setRenderMode('canvas');
    } else {
      setRenderMode('svg');
    }
  }, [data]);

  return renderMode === 'canvas' ? (
    <CanvasGraph data={data} />
  ) : (
    <SVGGraph data={data} />
  );
}
```

**Canvas 渲染器示例：**

```typescript
function CanvasGraph({ data }: GraphCanvasProps) {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const containerRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    const ctx = canvas.getContext('2d');
    if (!canvas || !ctx || !data) return;

    // 设置 Canvas 尺寸
    canvas.width = containerRef.current.clientWidth;
    canvas.height = containerRef.current.clientHeight;

    // 清除
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    // 绘制边
    data.edges.forEach(edge => {
      const source = data.nodes.find(n => n.id === edge.source);
      const target = data.nodes.find(n => n.id === edge.target);
      if (source && target && source.x && source.y && target.x && target.y) {
        ctx.beginPath();
        ctx.moveTo(source.x, source.y);
        ctx.lineTo(target.x, target.y);
        ctx.strokeStyle = '#ddd';
        ctx.lineWidth = 2;
        ctx.stroke();
      }
    });

    // 绘制节点
    data.nodes.forEach(node => {
      if (!node.x || !node.y) return;
      ctx.beginPath();
      ctx.arc(node.x, node.y, node.size || 30, 0, Math.PI * 2);
      ctx.fillStyle = node.color || '#3B82F6';
      ctx.fill();

      ctx.fillStyle = '#fff';
      ctx.textAlign = 'center';
      ctx.fillText(node.label, node.x, node.y + 45);
    });
  }, [data]);

  return (
    <div className="graph-canvas canvas-mode" ref={containerRef}>
      <canvas ref={canvasRef} />
    </div>
  );
}
```

### 1.2 增量更新

**问题：** 每次 AI 返回新图表都全量重绘

**解决方案：** 对比新旧数据，只更新变化的节点

```typescript
function useIncrementalUpdate(currentData: GraphData | null, newData: GraphData | null) {
  return useMemo(() => {
    if (!currentData || !newData) return newData;

    // 找出新增的节点
    const addedNodes = newData.nodes.filter(
      n => !currentData.nodes.find(c => c.id === n.id)
    );

    // 找出删除的节点
    const removedNodes = currentData.nodes.filter(
      n => !newData.nodes.find(c => c.id === n.id)
    );

    // 找出更新的节点
    const updatedNodes = newData.nodes.filter(n => {
      const old = currentData.nodes.find(c => c.id === n.id);
      return old && JSON.stringify(old) !== JSON.stringify(n);
    });

    return {
      addedNodes,
      removedNodes,
      updatedNodes,
      unchangedNodes: currentData.nodes.filter(n =>
        !addedNodes.includes(n) && !removedNodes.includes(n) && !updatedNodes.includes(n)
      )
    };
  }, [currentData, newData]);
}
```

### 1.3 防抖渲染

**问题：** 连续快速的状态更新导致频繁重绘

**解决方案：** 使用防抖合并渲染

```typescript
import { useDebouncedValue } from '../hooks/useDebouncedValue';

function GraphCanvas({ data }: GraphCanvasProps) {
  // 300ms 防抖
  const debouncedData = useDebouncedValue(data, 300);

  return <SVGGraph data={debouncedData} />;
}
```

## 三、布局优化

### 3.1 布局策略扩展

```typescript
// src/data/layoutStrategies.ts

interface LayoutStrategy {
  name: string;
  createSimulation: (nodes: any[], links: any[], width: number, height: number) => any;
  getNodePosition: (node: any, index: number, total: number) => { x: number; y: number };
}

export const layoutStrategies: Record<string, LayoutStrategy> = {
  // 力导向布局 - 通用
  force: {
    name: '力导向布局',
    createSimulation: (nodes, links, width, height) => {
      return d3.forceSimulation(nodes)
        .force('link', d3.forceLink(links).id((d: any) => d.id).distance(120))
        .force('charge', d3.forceManyBody().strength(-500))
        .force('center', d3.forceCenter(width / 2, height / 2));
    }
  },

  // 树形布局 - 层级结构
  tree: {
    name: '树形布局',
    createSimulation: (nodes, links, width, height) => {
      const root = d3.stratify()
        .id((d: any) => d.id)
        .parentId((d: any) => {
          const edge = links.find(l => l.target === d.id);
          return edge?.source;
        })(nodes);

      return d3.tree().size([width - 100, height - 100])(root);
    }
  },

  // 时间轴布局 - 阶段式路线图
  timeline: {
    name: '时间轴布局',
    createSimulation: (nodes, links, width, height) => {
      // 按 level 分组，水平排列
      const levelMap = new Map<number, any[]>();
      nodes.forEach((n: any, i: number) => {
        const level = n.level || 0;
        if (!levelMap.has(level)) levelMap.set(level, []);
        levelMap.get(level)!.push(n);
      });

      // 设置固定位置
      levelMap.forEach((levelNodes, level) => {
        const x = (level + 1) * (width / (levelMap.size + 1));
        levelNodes.forEach((node: any, i: number) => {
          node.x = x;
          node.y = (i + 1) * (height / (levelNodes.length + 1));
        });
      });

      return d3.forceSimulation(nodes)
        .force('x', d3.forceX((d: any) => d.x).strength(0.9))
        .force('y', d3.forceY((d: any) => d.y).strength(0.9));
    }
  },

  // 网格布局 - 分类展示
  grid: {
    name: '网格布局',
    createSimulation: (nodes, links, width, height) => {
      const cols = Math.ceil(Math.sqrt(nodes.length));
      const cellWidth = width / cols;
      const cellHeight = height / Math.ceil(nodes.length / cols);

      nodes.forEach((n: any, i: number) => {
        n.x = (i % cols) * cellWidth + cellWidth / 2;
        n.y = Math.floor(i / cols) * cellHeight + cellHeight / 2;
      });

      return d3.forceSimulation(nodes)
        .force('x', d3.forceX((d: any) => d.x).strength(0.9))
        .force('y', d3.forceY((d: any) => d.y).strength(0.9));
    }
  },

  // 环形布局 - 循环关系
  circular: {
    name: '环形布局',
    createSimulation: (nodes, links, width, height) => {
      const radius = Math.min(width, height) / 2 - 50;
      const angleStep = (2 * Math.PI) / nodes.length;

      nodes.forEach((n: any, i: number) => {
        n.x = width / 2 + radius * Math.cos(i * angleStep);
        n.y = height / 2 + radius * Math.sin(i * angleStep);
      });

      return d3.forceSimulation(nodes)
        .force('x', d3.forceX(width / 2).strength(0.1))
        .force('y', d3.forceY(height / 2).strength(0.1));
    }
  }
};
```

### 3.2 自动布局选择

AI 返回数据时可以包含 `layoutHint`，前端根据 Hint 选择最优布局：

```typescript
function selectBestLayout(data: GraphData): string {
  // 如果 AI 已经指定了 layout，优先使用
  if (data.layout) return data.layout;

  // 根据图表类型和结构自动选择
  const nodeCount = data.nodes.length;
  const hasLevels = data.nodes.some(n => n.level !== undefined);
  const isTreeStructure = data.edges.every(e =>
    data.edges.filter(ee => ee.target === e.target).length <= 1
  );

  if (hasLevels) return 'timeline';
  if (isTreeStructure && nodeCount < 50) return 'tree';
  if (nodeCount > 100) return 'grid';

  return 'force';
}
```

### 4.1 导出功能

```typescript
async function exportAsPNG() {
  const svg = document.querySelector('.graph-svg');
  if (!svg) return;

  const canvas = document.createElement('canvas');
  const ctx = canvas.getContext('2d');
  const svgData = new XMLSerializer().serializeToString(svg);
  const img = new Image();

  img.onload = () => {
    canvas.width = img.width;
    canvas.height = img.height;
    ctx.drawImage(img, 0, 0);
    const png = canvas.toDataURL('image/png');

    const link = document.createElement('a');
    link.download = 'chart.png';
    link.href = png;
    link.click();
  };

  img.src = 'data:image/svg+xml;base64,' + btoa(svgData);
}
```
