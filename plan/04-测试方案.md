```typescript
// src/__tests__/stores/graphStore.test.ts
import { renderHook, act } from '@testing-library/react';
import { useGraphStore } from '../stores/graphStore';

describe('graphStore', () => {

  describe('setGraphData', () => {
    it('应该正确存储图表数据', () => {
      const { result } = renderHook(() => useGraphStore());

      const testData = {
        type: 'roadmap',
        title: 'React 学习路线图',
        nodes: [
          { id: '1', label: 'HTML/CSS', color: '#64748B' },
          { id: '2', label: 'JavaScript', color: '#F59E0B' },
          { id: '3', label: 'React', color: '#3B82F6' },
        ],
        edges: [
          { source: '1', target: '2' },
          { source: '2', target: '3' },
        ],
        layout: 'timeline',
      };

      act(() => {
        result.current.setGraphData(testData);
      });

      expect(result.current.graphData).toEqual(testData);
      expect(result.current.graphData?.nodes).toHaveLength(3);
    });

    it('应该能存储不同类型的图表数据', () => {
      const { result } = renderHook(() => useGraphStore());

      const flowchartData = {
        type: 'flowchart',
        title: '登录流程',
        nodes: [
          { id: '1', label: '输入账号' },
          { id: '2', label: '输入密码' },
          { id: '3', label: '点击登录' },
        ],
        edges: [
          { source: '1', target: '2' },
          { source: '2', target: '3' },
        ],
        layout: 'force',
      };

      act(() => {
        result.current.setGraphData(flowchartData);
      });

      expect(result.current.graphData?.type).toBe('flowchart');
    });
  });

  describe('clearGraph', () => {
    it('应该正确清空图表数据', () => {
      const { result } = renderHook(() => useGraphStore());

      act(() => {
        result.current.setGraphData({
          type: 'roadmap',
          title: '测试',
          nodes: [],
          edges: [],
          layout: 'force',
        });
      });

      act(() => {
        result.current.clearGraph();
      });

      expect(result.current.graphData).toBeNull();
    });
  });

  describe('setLoading', () => {
    it('应该正确设置加载状态', () => {
      const { result } = renderHook(() => useGraphStore());

      expect(result.current.isLoading).toBe(false);

      act(() => {
        result.current.setLoading(true);
      });

      expect(result.current.isLoading).toBe(true);
    });
  });
});
```

```typescript
// src/__tests__/stores/chatStore.test.ts
import { renderHook, act } from '@testing-library/react';
import { useChatStore } from '../stores/chatStore';

describe('chatStore', () => {

  describe('addMessage', () => {
    it('应该正确添加用户消息', () => {
      const { result } = renderHook(() => useChatStore());

      act(() => {
        result.current.addMessage({
          role: 'user',
          content: '帮我画一个路线图'
        });
      });

      expect(result.current.messages).toHaveLength(1);
      expect(result.current.messages[0].role).toBe('user');
      expect(result.current.messages[0].content).toBe('帮我画一个路线图');
      expect(result.current.messages[0].id).toBeDefined();
      expect(result.current.messages[0].timestamp).toBeDefined();
    });

    it('应该正确添加 AI 消息', () => {
      const { result } = renderHook(() => useChatStore());

      act(() => {
        result.current.addMessage({
          role: 'assistant',
          content: '已为你生成图表',
          hasGraph: true
        });
      });

      expect(result.current.messages[0].hasGraph).toBe(true);
    });

    it('应该保留之前的历史消息', () => {
      const { result } = renderHook(() => useChatStore());

      act(() => {
        result.current.addMessage({ role: 'user', content: '消息1' });
      });
      act(() => {
        result.current.addMessage({ role: 'assistant', content: '回复1' });
      });
      act(() => {
        result.current.addMessage({ role: 'user', content: '消息2' });
      });

      expect(result.current.messages).toHaveLength(3);
    });
  });

  describe('setSessionId', () => {
    it('应该正确设置会话 ID', () => {
      const { result } = renderHook(() => useChatStore());

      act(() => {
        result.current.setSessionId('session-123');
      });

      expect(result.current.sessionId).toBe('session-123');
    });
  });

  describe('clearMessages', () => {
    it('应该清空所有消息', () => {
      const { result } = renderHook(() => useChatStore());

      act(() => {
        result.current.addMessage({ role: 'user', content: '测试' });
      });

      expect(result.current.messages).toHaveLength(1);

      act(() => {
        result.current.clearMessages();
      });

      expect(result.current.messages).toHaveLength(0);
    });
  });
});
```

```typescript
// src/__tests__/components/AISidePanel.test.tsx
import { render, screen, fireEvent, waitFor } from '@testing-library/react';
import { AISidePanel } from '../../pages/Graph/AISidePanel';
import { useGraphStore } from '../../stores/graphStore';

// Mock API
jest.mock('../../api/ai', () => ({
  chatWithAI: jest.fn(),
  createSession: jest.fn(),
}));

describe('AISidePanel', () => {
  beforeEach(() => {
    jest.clearAllMocks();
  });

  describe('发送消息', () => {
    it('应该能够发送消息并显示在列表中', async () => {
      const mockChat = require('../../api/ai').chatWithAI;
      mockChat.mockResolvedValue({
        data: { message: '这是 AI 回复' }
      });

      const mockCreateSession = require('../../api/ai').createSession;
      mockCreateSession.mockResolvedValue({
        data: { sessionId: 'test-session' }
      });

      render(<AISidePanel onGraphUpdate={jest.fn()} />);

      // 等待会话创建
      await waitFor(() => {
        expect(mockCreateSession).toHaveBeenCalled();
      });

      // 输入消息
      const input = screen.getByPlaceholderText('描述你想要的技术图表...');
      fireEvent.change(input, { target: { value: '测试消息' } });

      // 发送
      const sendBtn = screen.getByRole('button', { name: /发送/i });
      fireEvent.click(sendBtn);

      // 验证消息显示
      await waitFor(() => {
        expect(screen.getByText('测试消息')).toBeInTheDocument();
      });
    });

    it('AI 返回图表数据时应该更新图表', async () => {
      const mockChat = require('../../api/ai').chatWithAI;
      const graphData = {
        type: 'roadmap',
        title: '测试图表',
        nodes: [{ id: '1', label: 'Node 1' }],
        edges: [],
        layout: 'force',
      };
      mockChat.mockResolvedValue({
        data: {
          message: '已生成图表',
          graphData
        }
      });

      const onGraphUpdate = jest.fn();
      render(<AISidePanel onGraphUpdate={onGraphUpdate} />);

      // 触发发送...
      await waitFor(() => {
        expect(onGraphUpdate).toHaveBeenCalledWith(graphData);
      });
    });

    it('发送失败时应该显示错误消息', async () => {
      const mockChat = require('../../api/ai').chatWithAI;
      mockChat.mockRejectedValue(new Error('网络错误'));

      const mockCreateSession = require('../../api/ai').createSession;
      mockCreateSession.mockResolvedValue({
        data: { sessionId: 'test-session' }
      });

      render(<AISidePanel onGraphUpdate={jest.fn()} />);

      // 等待初始化
      await waitFor(() => {
        expect(mockCreateSession).toHaveBeenCalled();
      });

      // 发送消息触发错误
      const input = screen.getByPlaceholderText('描述你想要的技术图表...');
      fireEvent.change(input, { target: { value: '测试' } });
      fireEvent.click(screen.getByRole('button', { name: /发送/i }));

      await waitFor(() => {
        expect(screen.getByText(/抱歉/)).toBeInTheDocument();
      });
    });
  });
});
```

```typescript
// src/__tests__/components/GraphCanvas.test.tsx
import { render, screen } from '@testing-library/react';
import { GraphCanvas } from '../../pages/Graph/GraphCanvas';
import type { GraphData } from '../../types/graph';

describe('GraphCanvas', () => {

  const mockGraphData: GraphData = {
    type: 'roadmap',
    title: '测试图表',
    nodes: [
      { id: '1', label: 'Node 1', color: '#3B82F6' },
      { id: '2', label: 'Node 2', color: '#10B981' },
    ],
    edges: [
      { source: '1', target: '2' },
    ],
    layout: 'force',
  };

  it('应该在有数据时渲染 SVG', () => {
    render(<GraphCanvas data={mockGraphData} />);

    const svg = screen.getByRole('img', { hidden: true });
    expect(svg).toBeInTheDocument();
  });

  it('应该在无数据时显示空状态', () => {
    render(<GraphCanvas data={null} />);

    expect(screen.getByText(/让 AI 为你生成图表/)).toBeInTheDocument();
  });

  it('应该渲染正确数量的节点', () => {
    render(<GraphCanvas data={mockGraphData} />);

    // D3 会创建 g.node 元素
    const nodes = document.querySelectorAll('.node');
    expect(nodes).toHaveLength(2);
  });

  it('应该根据不同 layout 应用不同的渲染策略', () => {
    const timelineData = {
      ...mockGraphData,
      layout: 'timeline' as const,
    };

    const { rerender } = render(<GraphCanvas data={mockGraphData} />);

    // 切换到 timeline 布局
    rerender(<GraphCanvas data={timelineData} />);

    // 验证渲染正常
    const svg = screen.getByRole('img', { hidden: true });
    expect(svg).toBeInTheDocument();
  });
});
```

```typescript
// e2e/graph-ai.spec.ts
import { test, expect } from '@playwright/test';

test.describe('AI 图表生成', () => {

  test.beforeEach(async ({ page }) => {
    // 登录
    await page.goto('/login');
    await page.fill('[placeholder="用户名"]', 'testuser');
    await page.fill('[placeholder="密码"]', 'password123');
    await page.click('button[type="submit"]');

    // 等待登录完成
    await page.waitForURL('/');

    // 跳转到图谱页面
    await page.goto('/graph');
  });

  test('完整流程：发送需求 → AI 生成 → 图表渲染', async ({ page }) => {
    // 验证 AI 对话面板存在
    await expect(page.locator('.ai-side-panel')).toBeVisible();

    // 输入图表需求
    const input = page.locator('.message-input');
    await input.fill('帮我画一个前端学习路线图，包含 HTML、CSS、JavaScript、React');

    // 点击发送
    await page.click('.send-btn');

    // 等待 AI 回复
    await expect(page.locator('.message.assistant')).toBeVisible({
      timeout: 10000
    });

    // 验证图表已生成（SVG 中有节点）
    await expect(page.locator('.graph-canvas svg .node').first()).toBeVisible({
      timeout: 10000
    });

    // 验证节点数量
    const nodeCount = await page.locator('.graph-canvas svg .node').count();
    expect(nodeCount).toBeGreaterThan(0);
  });

  test('修改图表：发送修改需求 → AI 更新 → 图表更新', async ({ page }) => {
    // 先生成一个图表
    await page.locator('.message-input').fill('画一个前端路线图');
    await page.click('.send-btn');
    await page.waitForSelector('.graph-canvas svg .node', { timeout: 10000 });
```

```typescript
// e2e/error-handling.spec.ts
import { test, expect } from '@playwright/test';

test.describe('错误处理', () => {

  test('网络错误时应该显示友好提示', async ({ page }) => {
    await page.goto('/graph');

    // 使用代理模拟网络错误

    await page.locator('.message-input').fill('测试');

        })
      });
    });

    await page.locator('.message-input').fill('测试');
    await page.click('.send-btn');

    await expect(page.locator('.message.assistant')).toContainText('超时');
```

```typescript
import { test, expect } from '@playwright/test';
```
